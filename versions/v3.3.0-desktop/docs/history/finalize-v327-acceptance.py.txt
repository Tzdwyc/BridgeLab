"""Seal the two target fixes without concealing the retained live failure."""
import hashlib
import json
import sys
from pathlib import Path
from zipfile import ZipFile

root = Path(__file__).resolve().parents[1]
folder = root / 'artifacts/v327-two-fixes-20261108'
mobile = root.parent / 'v3.2.7-mobile'
read = lambda p: json.loads(p.read_text(encoding='utf-8-sig'))
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
write = lambda p, v: p.write_text(json.dumps(v, ensure_ascii=False, indent=2), encoding='utf-8')
base = read(folder / 'BASELINE.json')
for name, digest in base['files'].items():
    assert sha(root.parent / 'v3.2.6-desktop' / name) == digest, name
old_zip = root.parent / 'v3.2.6-desktop/release/BridgeLab-v3.2.6-desktop.zip'
old_apk = root.parent / 'v3.2.6-mobile/release/BridgeLab-v3.2.6-mobile-android.apk'
assert sha(old_zip) == base['desktopArchiveSHA256']
assert sha(old_apk) == base['mobileArchiveSHA256']
units = read(folder / 'final-unit-results.json')
assert units['numPassedTests'] == 375 and units['numFailedTests'] == 0
actual = read(folder / 'actual-table-001.json')
summary = read(folder / 'acceptance-summary.json')
assert summary['sourceEvidenceSHA256'] == sha(folder / 'actual-table-001.json')
assert actual['fatalError'] is None and actual['inputsPreserved'] and actual['sourceStableDuringRun']
assert actual['training'] == actual['newMotherDeals'] == actual['playAdvisorCalls'] == actual['ddsCalls'] == 0
assert actual['completedControllerQueries'] == 44 and actual['actualActions'] == 67
assert actual['completedApprovedFixQueries'] == actual['completedLanActualPollQueries'] == 4
assert summary['approvedFixControllerPassed'] == summary['approvedFixLanPassed'] == 4
assert summary['allFortyFrozenV326ObservationsQualified']
assert summary['priorFiveTargetRegressionPassed'] == 10 and summary['originalSlAndPrecisionControlsPassed']
targets = [r for r in actual['rows'] if r['group'] == 'two-approved-fixes']
assert len(targets) == 4
for row in targets + actual['lanRows']:
    assert row['publicMeaning']['defined'] and row['qualification']['allowed'] and not row['violations']
    assert row['actualApplyActionAccepted'] and row['engineIntegrityPassed'] and row['engineReplayPassed']
unresolved = summary['unresolvedRows']
assert len(unresolved) == 2 and all(r['id'] == 'enemy-natural-diamonds-not-our-second-suit' for r in unresolved)
assert {r['system'] for r in unresolved} == {'sayc', 'two-over-one'}
assert all(r['actual']['kind'] == 'pass' and 'unfulfilled-forcing-obligation' in r['reasons'] for r in unresolved)
assert summary['overallTestPassed'] is False and summary['overallTestExitCode'] == 1
assert len([r for r in actual['rows'] if r['qualification']['allowed']]) == 42
for name, digest in actual['sourceHashes'].items():
    assert sha(root / name) == digest, name
fixed = read(folder / 'fixed33-final/fixed33-actual-table.json')
prior = read(root / 'artifacts/v326-three-flows-20261108/fixed33-final/fixed33-actual-table.json')
prior_rows = {(r['pointId'], r['system']): r for r in prior['rows']}
assert all(r['newCall'] == prior_rows[r['pointId'], r['system']]['newCall'] for r in fixed['rows'])
assert fixed['validExecution'] and fixed['actualSingleBidQueriesCompleted'] == 66
assert fixed['protectedCallRows'] == fixed['originalSLRollback']['completedQueries'] == 18
assert all(r['frozenOriginalCallPreserved'] for r in fixed['originalSLRollback']['rows'])
inherited = read(folder / 'inherited-text-comparison.json')
assert inherited['sameSixInheritedFailures'] and not inherited['newFailures']
ui = read(folder / 'ui-final/verification.json')
delivery = read(root / 'artifacts/delivery-verification-v3.2.7.json')
assert ui['passed'] and len(ui['results']) == 24
assert ui['sha256'] == delivery['mainBundleSHA256'] == sha(root / 'dist/assets' / ui['bundle'])
assert all((root / p).stat().st_mtime <= (root / 'dist/assets' / ui['bundle']).stat().st_mtime
           for p in delivery['actualSourceChanges'])
result = {
    'version': '3.2.7', 'androidVersionCode': 67, 'baseVersion': '3.2.6',
    'status': 'two approved fixes verified; additional live counterexample remains unresolved',
    'approvedFixTargetsPassed': True, 'overallLiveRegressionPassed': False,
    'overallLiveRegressionExitCode': 1, 'unresolvedLiveRows': unresolved,
    'trainingRun': False, 'newMotherDealPerformanceCohort': False,
    'mainBundle': ui['bundle'], 'mainBundleSHA256': ui['sha256'],
    'targetedUnitTests': {'passed': 375, 'failed': 0, 'files': 19},
    'inheritedLegacyTextAssertions': {'sameFailuresAsV326': 6, 'newFailures': 0},
    'targetPositions': 2, 'actualTargetControllerCalls': 4, 'actualTargetLanPollCalls': 4,
    'mainControllerCalls': 44, 'mainControllerQualified': 42, 'totalActualActions': 67,
    'priorFortyObservationsAllQualified': True, 'priorFiveTargetRegressionCalls': 10,
    'entrypointDifference': summary['entrypointDifference'],
    'fixed33': {'actualCalls': 66, 'callsChangedFromV326': 0,
                'protectedRowsUnchanged': 18, 'originalSLRollbackMatched': 18},
    'productionPhoneViewportChecks': 24, 'defaultModel': 'seed101-epoch08',
    'rollbackModel': 'original-sl', 'lookaheadDisabled': True,
    'publicAgreementVersionUnchanged': 'v2.4',
    'sourceChanges': delivery['actualSourceChanges'], 'sourceSHA256': delivery['sourceSHA256'],
    'oldV326SourceAndPackagesPreserved': True,
    'physicalPhoneInstallation': False, 'emulatorInstallation': False,
    'realMultiDeviceLanValidation': False, 'wholeBoardPerformanceClaim': False,
    'actualEvidence': 'actual-table-001.json', 'acceptanceSummary': 'acceptance-summary.json',
}
if '--prepackage' in sys.argv:
    result['packageHashes'] = 'Pending in external FINAL_ACCEPTANCE.json'
    write(folder / 'PREPACKAGE_ACCEPTANCE.json', result)
    print('Target checks verified; retained live failure recorded. 375 unit checks, 8 target actions, 24 UI checks.')
    raise SystemExit(0)
desktop = read(root / 'release/DESKTOP_PACKAGE_VERIFICATION_v3.2.7.json')
assert sha(Path(desktop['archive'])) == desktop['sha256']
validation = read(mobile / 'docs/VALIDATION_v3.2.7.json')
upgrade = read(mobile / 'docs/UPGRADE_SIGNATURE_VALIDATION_v3.2.7.json')
apk = mobile / 'release/BridgeLab-v3.2.7-mobile-android.apk'
assert sha(apk) == validation['sha256'] == upgrade['current']['sha256']
assert validation['mainBundleHashMatch'] and validation['unchangedDistFileHashesMatch']
assert validation['desktopMainBundleSha256'] == ui['sha256'] and validation['desktopDistFileCount'] == 29
assert upgrade['samePublicCertificate'] and upgrade['current']['versionCode'] == 67
assert upgrade['previous']['versionCode'] == 66
models = []
with ZipFile(apk) as new, ZipFile(old_apk) as old:
    for name in sorted(n for n in new.namelist() if n.startswith('assets/flutter_assets/assets/www/models/') and not n.endswith('/')):
        assert new.read(name) == old.read(name), name
        models.append({'asset': name, 'sha256': hashlib.sha256(new.read(name)).hexdigest(), 'unchangedFromV326': True})
assert len(models) == 9
result.update(desktopPackage=desktop, androidPackage={
    'apk': str(apk), 'bytes': apk.stat().st_size, 'sha256': sha(apk),
    'all29DistResourcesMatch': True, 'sameApplicationId': upgrade['sameApplicationId'],
    'sameCertificate': True, 'certificateSHA256': upgrade['current']['publicCertificateSha256']},
    all9ModelAssets=models,
    documentationSnapshotNote='Package hashes are external. Final production source and executable/model assets match; post-build receipts do not change executable content.')
write(folder / 'FINAL_ACCEPTANCE.json', result)
write(mobile / 'docs/V327_RELEASE_VERIFICATION.json', result)
print(json.dumps({'deliveryVerificationComplete': True, 'approvedFixTargetsPassed': True,
                  'overallLiveRegressionPassed': False, 'retainedLiveFailures': 2,
                  'desktopSHA256': desktop['sha256'], 'apkSHA256': sha(apk)}))
